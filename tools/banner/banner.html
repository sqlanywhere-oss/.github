-- Watcom-SQL and Transact-SQL, one parserCREATE PROCEDURE recent_orders( IN days INT DEFAULT 7 )RESULT ( order_id INT, total NUMERIC(12,2) )BEGIN  SELECT o.order_id, SUM( l.qty * l.price )    FROM orders o KEY JOIN order_lines l   WHERE o.placed >= DATEADD( day, -days, CURRENT DATE )   GROUP BY o.order_id   ORDER BY 2 DESC;END;
